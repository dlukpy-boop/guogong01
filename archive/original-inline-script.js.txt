(()=>{
const st=document.getElementById('stage');const prev=document.getElementById('prev'),next=document.getElementById('next');
const steps=[{s:'cover',r:0},{s:'overview',r:0},{s:'history',r:1},{s:'history',r:2},{s:'history',r:3},{s:'history1912',r:1},{s:'history1912',r:2},{s:'history1912',r:3}];let ix=0;
function fit(){let f=Math.min(innerWidth/1600,innerHeight/900);st.style.transform=`translate(-50%,-50%) scale(${f})`;}
function go(n){ix=Math.min(steps.length-1,Math.max(0,n));const x=steps[ix];st.dataset.scene=x.s;st.dataset.reveal=x.r;prev.disabled=ix===0;next.disabled=ix===steps.length-1;next.title=next.disabled?'下一站1914尚未製作':'顯示下一個畫面';prev.title=prev.disabled?'目前是封面':'回到上一個畫面';document.getElementById('scene-indicator').textContent=ix<2?'國共合作 1.0':`國民黨沿革 · ${x.s==='history'?'1905':'1912'} (${x.r}/3)`;document.querySelectorAll('#timeline .t').forEach((el,i)=>el.classList.toggle('active',i===(x.s==='history1912'?1:0)));}
prev.addEventListener('click',()=>go(ix-1));next.addEventListener('click',()=>go(ix+1));document.getElementById('home').addEventListener('click',()=>go(0));document.getElementById('choiceK').addEventListener('click',()=>go(2));
addEventListener('keydown',e=>{if(['ArrowRight','PageDown',' '].includes(e.key)){e.preventDefault();go(ix+1);}else if(['ArrowLeft','PageUp'].includes(e.key)){e.preventDefault();go(ix-1);}else if(e.key==='Home'||e.key==='Escape'){e.preventDefault();go(0);}});
addEventListener('resize',fit);fit();go(0);
})();
